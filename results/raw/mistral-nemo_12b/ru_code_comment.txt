"""
Нормализует российский номер телефона в произвольном виде и возвращает его в каноническом виде '+79123456789'.
Если из строки нельзя получить корректный российский номер (11 цифр, начинающихся с 7 или 8, либо 10 цифр без кода страны) — вернуть None.
"""
def нормализовать_телефон(s: str) -> str | None:
    s = ''.join(filter(str.isdigit, s))
    if len(s) == 10 or (len(s) == 11 and s[0] in ('7', '8')):
        return f"+7{s}"
