def нормализовать_телефон(s: str) -> str | None:
    """Функция нормализует российский номер телефона в канонический формат '+7ХХХХХХХХХ'.
    
    Параметры:
        s: строка с номером телефона
        
    Возвращает:
        канонический номер или None, если номер некорректен
    """
    digits = ''.join(filter(str.isdigit, s))
    if len(digits) == 11 and digits[0] in '78':
        if digits[0] == '8':
            digits = '7' + digits[1:]
        return '+' + digits
    elif len(digits) == 10:
        return '+7' + digits
    else:
        return None
