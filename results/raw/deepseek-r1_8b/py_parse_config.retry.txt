def parse_config(text: str) -> dict:
    import re
    lines = text.splitlines()
    result = {}
    current_section = None
    first_section_started = False
    for line in lines:
        line = line.strip()
        if not line or line.startswith('#') or line.startswith(';'):
            continue
        if line.startswith('[') and line.endswith(']'):
            section_name = line[1:-1].strip()
            if not first_section_started:
                first_section_started = True
                current_section = section_name
            else:
                if current_section is not None:
                    result[current_section] = {}
                current_section = section_name
        else:
            if '=' not in line:
                continue
            key, value_str = line.split('=', 1)
            key = key.strip()
            if first_section_started and current_section is None:
                section_name = ''
            else:
                section_name = current_section
            if value_str.startswith('"') and value_str.endswith('"'):
                value = value_str[1:-1]
            else:
                try:
                    if value_str.lower() == 'true':
                        value = True
                    elif value_str.lower() == 'false':
                        value = False
                    elif value_str.isdigit():
                        value = int(value_str)
                    elif '.' in value_str and value_str.replace('.', '', 1).isdigit():
                        value = float(value_str)
                    else:
                        value = value_str
                except:
                    value = value_str
            if section_name not in result:
                result[section_name] = {}
            result[section_name][key] = value
    if first_section_started and current_section is not None:
        result[current_section] = {}
    return result
