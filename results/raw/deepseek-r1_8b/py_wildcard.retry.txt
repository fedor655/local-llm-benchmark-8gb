def is_match(s: str, p: str) -> bool:
    i = 0
    j = 0
    while i < len(s) and j < len(p):
        if p[j] == '*':
            j += 1
            continue
        elif p[j] == '?':
            i += 1
            j += 1
        else:
            if i >= len(s) or s[i] != p[j]:
                break
            else:
                i += 1
                j += 1
    return i == len(s) and j == len(p)
