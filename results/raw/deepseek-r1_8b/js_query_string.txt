  function parseQuery(qs) {
    // Убираем ведущий '?'
    if (qs.startsWith('?')) {
      qs = qs.substring(1);
    }

    // Разбиваем на части, игнорируя пустые строки
    let parts = qs.split('&').filter(part => part !== '');

    let result = {};

    for (let part of parts) {
      // Проверяем, есть ли '=' в части
      let indexOfEqual = part.indexOf('=');
      let key, value;

      if (indexOfEqual === -1) {
        // ключ без значения
        key = decodeURIComponent(part);
        value = '';
      } else {
        key = decodeURIComponent(part.substring(0, indexOfEqual));
        value = decodeURIComponent(part.substring(indexOfEqual + 1));
      }

      // Если ключ уже есть в объекте, то добавляем значение в массив
      if (result.hasOwnProperty(key)) {
        // Проверяем, является ли текущее значение массивом
        if (Array.isArray(result[key])) {
          result[key].push(value);
        } else {
          result[key] = [result[key], value];
        }
      } else {
        result[key] = value;
      }
    }

    return result;
  }

```javascript
function parseQuery(qs) {
    if (qs.startsWith('?')) qs = qs.substring(1);
    const parts = qs.split('&').filter(part => part !== '');
    const result = {};
    for (const part of parts) {
        const [key, value] = part.includes('=') ? 
            [part.substring(0, part.indexOf('=')), part.substring(part.indexOf('=') + 1)] : 
            [part, ''];
        const decodedKey = decodeURIComponent(key);
        const decodedValue = value ? decodeURIComponent(value) : '';
        if (result.hasOwnProperty(decodedKey)) {
            if (!Array.isArray(result[decodedKey])) result[decodedKey] = [result[(decodedKey))];
            result[decodedKey].push(decodedValue);
        } else {
            result[decodedKey] = decodedValue;
        }
    }
    return result;
}
```
