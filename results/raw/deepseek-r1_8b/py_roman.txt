def int_to_roman(n: int) -> str:
    val = [
        (1000, 'M'),
        (900, 'CM'),
        (500, 'D'),
        (400, 'CD'),
        (100, 'C'),
        (90, 'XC'),
        (50, 'L'),
        (40, 'XL'),
        (10, 'X'),
        (9, 'IX'),
        (5, 'V'),
        (4, 'IV'),
        (1, 'I')
    ]
    roman = ''
    for value, numeral in val:
        while n >= value:
            roman += numeral
            n -= value
    return roman

def roman_to_int(s: str) -> int:
    roman_val = {'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000}
    total = 0
    prev = 0
    for char in reversed(s):
        current = roman_val[char]
        if current < prev:
            total -= current
        else:
            total += current
        prev = current
    return total
