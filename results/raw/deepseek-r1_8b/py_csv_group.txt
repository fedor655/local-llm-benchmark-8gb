```python
import csv
from collections import defaultdict

def top_by_group(csv_text: str, group_col: str, value_col: str, k: int) -> list[tuple[str, float]]:
    lines = csv_text.splitlines()
    while lines and not lines[-1].strip():
        lines.pop()
    
    header = lines[0].split(',') if lines else []
    group_index = header.index(group_col) if group_col in header else -1
    value_index = header.index(value_col) if value_col in header else -1
    
    if group_index == -1 or value_index == -1:
        return []
    
    groups = defaultdict(lambda: {'sum': 0.0, 'valid': False})
    
    for line in lines[1:]:
        row = line.split(',')
        if len(row) <= max(group_index, value_index):
            continue
        try:
            group_val = row[group_index]
            value_val = float(row[value_index])
            if groups[group_val]['valid']:
                groups[group_val]['sum'] += value_val
            else:
                groups[group_val]['sum'] = value_val
                groups[group_val]['valid'] = True
        except ValueError:
            continue
    
    valid_groups = [(group, data['sum']) for group, data in groups.items() if data['valid']]
    valid_groups.sort(key=lambda x: (-x[1], x[0]))
    
    return valid_groups[:k]
```
