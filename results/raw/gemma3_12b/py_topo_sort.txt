def topo_sort(n: int, edges: list[tuple[int, int]]) -> list[int] | None:
    in_degree = [0] * n
    graph = [[] for _ in range(n)]
    for a, b in edges:
        graph[a].append(b)
        in_degree[b] += 1

    queue = [i for i in range(n) if in_degree[i] == 0]
    result = []
    count = 0

    while queue:
        u = queue.pop(0)
        result.append(u)
        count += 1
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)

    if count != n:
        return None
    else:
        return result
