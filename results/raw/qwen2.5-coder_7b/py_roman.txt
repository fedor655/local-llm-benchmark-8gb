def int_to_roman(n: int) -> str:
    val = [
        1000, 900, 500, 400,
        100, 90, 50, 40,
        10, 9, 5, 4,
        1
    ]
    syms = [
        "M", "CM", "D", "CD",
        "C", "XC", "L", "XL",
        "X", "IX", "V", "IV",
        "I"
    ]
    roman_num = ''
    i = 0
    while n > 0:
        for _ in range(n // val[i]):
            roman_num += syms[i]
            n -= val[i]
        i += 1
    return roman_num

def roman_to_int(s: str) -> int:
    val = {
        "I": 1,
        "V": 5,
        "X": 10,
        "L": 50,
        "C": 100,
        "D": 500,
        "M": 1000
    }
    int_num = 0
    i = 0
    while i < len(s):
        if i + 1 < len(s) and val[s[i]] < val[s[i + 1]]:
            int_num += val[s[i + 1]] - val[s[i]]
            i += 2
        else:
            int_num += val[s[i]]
            i += 1
    return int_num
