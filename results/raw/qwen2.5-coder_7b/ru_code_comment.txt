def нормализовать_телефон(s: str) -> str | None:
    """
    Нормализует российский номер телефона в каноническом виде '+79123456789'.
    Если из строки нельзя получить корректный российский номер, возвращает None.
    """
    s = ''.join(filter(str.isdigit, s))
    if len(s) == 10:
        return f'+7{s}'
    elif len(s) == 11 and (s[0] == '7' or s[0] == '8'):
        return '+' + s
    else:
        return None
