```python
import csv
from collections import defaultdict

def top_by_group(csv_text: str, group_col: str, value_col: str, k: int) -> list[tuple[str, float]]:
    groups = defaultdict(float)
    
    for row in csv.reader(csv_text.splitlines()):
        if not row:
            continue
        
        try:
            group_name = row[group_col]
            value = float(row[value_col])
            groups[group_name] += value
        except (ValueError, IndexError):
            continue
    
    sorted_groups = sorted(groups.items(), key=lambda x: (-x[1], x[0]))
    
    return sorted_groups[:k]
```
