function deepEqual(a, b) {
  // NaN equals NaN
  if (typeof a === 'number' && typeof b === 'number') {
    if (Number.isNaN(a) && Number.isNaN(b)) return true;
    if (a === 0 && b === 0 && (1 / a) !== (1 / b)) return false; // +0 vs -0
    return a === b;
  }

  // Strict equality for other primitives
  if (a === b) {
    // Handle +0 vs -0 for non-number types? No, only numbers have signed zero.
    // But if a === b and they are numbers, we already handled above.
    // If they are other primitives, === is fine.
    // However, if a and b are both 0 (numbers), we already returned in the number block.
    // So if we reach here with a === b, it's safe.
    return true;
  }

  // If either is null or not an object, they are not equal
  if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') {
    return false;
  }

  // Date comparison
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() === b.getTime();
  }
  if (a instanceof Date || b instanceof Date) {
    return false;
  }

  // Array vs Object check
  const aIsArray = Array.isArray(a);
  const bIsArray = Array.isArray(b);
  if (aIsArray !== bIsArray) {
    return false;
  }

  if (aIsArray) {
    if (a.length !== b.length) return false;
    for (let i = 0; i < a.length; i++) {
      if (!deepEqual(a[i], b[i])) return false;
    }
    return true;
  }

  // Plain objects
  const aKeys = Object.keys(a);
  const bKeys = Object.keys(b);
  if (aKeys.length !== bKeys.length) return false;

  for (const key of aKeys) {
    if (!Object.prototype.hasOwnProperty.call(b, key)) return false;
    if (!deepEqual(a[key], b[key])) return false;
  }

  return true;
}
