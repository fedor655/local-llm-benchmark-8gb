def нормализовать_телефон(s: str) -> str | None:
    """
    Нормализует российский номер телефона в канонический формат '+7XXXXXXXXXX'.
    
    Принимает номер в произвольном виде (с пробелами, дефисами, скобками,
    с кодом страны 7 или 8 или без него) и возвращает его в формате '+7XXXXXXXXXX'.
    Если номер не является корректным российским номером, возвращает None.
    
    Корректный номер:
    - 11 цифр, начинающихся с 7 или 8, или
    - 10 цифр (без кода страны), которые будут дополнены '7' в начале.
    
    Примеры:
    '+7 (912) 345-67-89' -> '+79123456789'
    '8-912-345-67-89' -> '+79123456789'
    '89123456789' -> '+79123456789'
    '9123456789' -> '+79123456789'
    """
    # Извлекаем только цифры
    digits = ''.join(c for c in s if c.isdigit())
    
    # Проверяем длину
    if len(digits) == 11:
        # Должен начинаться с 7 или 8
        if digits[0] in ('7', '8'):
            return '+7' + digits[1:]
        return None
    elif len(digits) == 10:
        # Без кода страны, добавляем 7
        return '+7' + digits
    else:
        return None
