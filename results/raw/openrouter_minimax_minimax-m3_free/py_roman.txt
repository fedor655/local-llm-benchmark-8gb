def int_to_roman(n: int) -> str:
    if not 1 <= n <= 3999:
        raise ValueError("n must be in 1..3999")
    vals = [
        (1000, "M"), (900, "CM"), (500, "D"), (400, "CD"),
        (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
        (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I"),
    ]
    out = []
    for v, sym in vals:
        while n >= v:
            out.append(sym)
            n -= v
    return "".join(out)


def roman_to_int(s: str) -> int:
    if not s:
        raise ValueError("empty string")
    values = {"I": 1, "V": 5, "X": 10, "L": 50, "C": 100, "D": 500, "M": 1000}
    pairs = {"IV": 4, "IX": 9, "XL": 40, "XC": 90, "CD": 400, "CM": 900}
    s = s.upper()
    i = 0
    total = 0
    n = len(s)
    while i < n:
        if i + 1 < n and s[i:i+2] in pairs:
            total += pairs[s[i:i+2]]
            i += 2
        else:
            if s[i] not in values:
                raise ValueError(f"invalid character: {s[i]}")
            total += values[s[i]]
            i += 1
    return total
