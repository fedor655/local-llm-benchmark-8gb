def нормализовать_телефон(s: str) -> str | None:
    """
    Нормализует российский номер телефона в стандартный формат '+7XXXXXXXXXX'.
    Возвращает None, если номер некорректный (не 11 цифр, начинающихся с 7 или 8, или 10 цифр без кода страны).
    """
    digits = ''.join(filter(str.isdigit, s))
    if len(digits) == 11 and digits[0] in ('7', '8'):
        return f'+7{digits[1:]}'
    elif len(digits) == 10 and digits[0] == '8':
        return f'+7{digits}'
    return None
