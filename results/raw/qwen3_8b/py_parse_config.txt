def parse_config(text: str) -> dict:
    result = {}
    current_section = None
    has_pairs_before_first_section = False
    lines = text.splitlines()
    
    for line in lines:
        line = line.strip()
        if not line or line.startswith('#') or line.startswith(';'):
            continue
        if line.startswith('['):
            section_name = line[1:-1].strip()
            if current_section is not None:
                result[current_section] = {}
            current_section = section_name
            if not current_section:
                current_section = ''
            if not result.get(current_section, None):
                result[current_section] = {}
            if not has_pairs_before_first_section and current_section == '':
                has_pairs_before_first_section = True
        else:
            key, value = line.split('=', 1)
            key = key.strip()
            value = value.strip()
            if value.startswith('"') and value.endswith('"'):
                value = value[1:-1]
            else:
                if value.lower() in ('true', 'false'):
                    value = value.lower() == 'true'
                elif value.isdigit():
                    value = int(value)
                elif value.replace('.', '', 1).isdigit():
                    value = float(value)
                else:
                    value = value
            if current_section is None:
                if '' not in result:
                    result[''] = {}
                result[''].update({key: value})
            else:
                result[current_section][key] = value
    
    if current_section is not None and '' in result:
        result.pop('')
    if has_pairs_before_first_section and '' not in result:
        result[''] = {}
    return result
