```python
def two_sum(nums: list[int], target: int) -> tuple[int, int] | None:
    num_map = {}
    for j in range(len(nums)):
        complement = target - nums[j]
        if complement in num_map:
            return (num_map[complement], j)
        num_map[nums[j]] = j
    return None
```
