def top_by_group(csv_text: str, group_col: str, value_col: str, k: int) -> list[tuple[str, float]]:
    rows = [line.strip() for line in csv_text.split('\n') if line.strip()]
    if not rows:
        return []
    headers = rows[0].split(',')
    group_idx = headers.index(group_col)
    value_idx = headers.index(value_col)
    groups = {}
    for row in rows[1:]:
        values = row.split(',')
        group_val = values[group_idx]
        try:
            val = float(values[value_idx])
            groups[group_val] = groups.get(group_val, 0) + val
        except (ValueError, IndexError):
            continue
    sorted_groups = sorted(groups.items(), key=lambda x: (-x[1], x[0]))
    return sorted_groups[:k]
