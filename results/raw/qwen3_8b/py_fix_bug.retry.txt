def search_range(nums, target):
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    first = lo if lo < len(nums) and nums[lo] == target else -1
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] > target:
            hi = mid - 1
        else:
            lo = mid + 1
    last = hi if hi >= 0 and nums[hi] == target else -1
    return (first, last)
