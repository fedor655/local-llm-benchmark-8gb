    if not intervals:

    intervals_sorted = sorted(intervals, key=lambda x: x[0])
    merged = [intervals_sorted[0]]
    for i in range(1, len(intervals_sorted)):
        current_interval = intervals_sorted[i]
        last_interval = merged[-1]
