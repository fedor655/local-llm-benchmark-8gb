Функция deepEqual(a, b) {

  // Проверка на примитивы
  if (a === b) {

  if (a !== a || b !== b) {
    return false;
  }

  // Проверка на массивы
  if (Array.isArray(a) && Array.isArray(b)) {
    if (a.length !== b.length) {
      return false;
    }
    for (let i = 0; i < a.length; i++) {
      if (!deepEqual(a[i], b[i])) {
        return false;
      }
    }
    return true;
  }

  // Проверка на объекты
  if (typeof a === 'object' && typeof b === 'object') {
    if (a === null || b === null) {
      return false;
    }
    if (Object.keys(a).length !== Object.keys(b).length) {
      return false;
    }
    for (const key in a) {

        return false;
      }
    }
    return true;
