def search_range(nums, target):
    # Возвращает (первый_индекс, последний_индекс) вхождения target
    # в отсортированном по возрастанию списке nums, либо (-1, -1).
    lo, hi = 0, len(nums) - 1
    first, last = -1, -1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            first = mid
            hi = mid - 1
        elif nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            last = mid
            lo = mid + 1
        elif nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return (first, last)
