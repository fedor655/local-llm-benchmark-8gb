def нормализовать_телефон(s: str) -> str | None:
    """
    Нормализует российский номер телефона.

    Args:
        s: Номер телефона в произвольном виде.

    Returns:
        Нормализованный номер телефона в каноническом виде '+79123456789' или None,
        если из строки нельзя получить корректный номер.
    """

    s = ''.join(filter(str.isdigit, s))

    if len(s) == 11 and (s.startswith('7') or s.startswith('8')):
        return '+' + s
    elif len(s) == 10:
        return '+7' + s
    else:
        return None
