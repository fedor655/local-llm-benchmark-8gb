def нормализовать_телефон(s: str) -> str | None:
    """
    Принимает российский номер телефона в произвольном виде и возвращает его в каноническом виде '+79123456789'.
    
    Если из строки нельзя получить корректный российский номер (11 цифр, начинающихся с 7 или 8, либо 10 цифр без кода страны) — вернуть None.
    
    :param s: Российский номер телефона в произвольном виде
    :return: Канонический номер телефона или None, если номер неверен
    """
    # Удалить все пробелы и знаки минуса из строки
    s = s.replace(' ', '').replace('-', '')
    
    # Если длина строки не равна 11 или 10 цифрам — вернуть None
    if len(s) not in (11, 10):
        return None
    
    # Если номер начинается с '8' и его длина больше 10 — вернуть None
    if s.startswith('8') and len(s) > 10:
        return None
    
    # Если номер начинается с '7' или '8' и его длина равна 11 — вернуть его в каноническом виде
    if (s.startswith('7') or s.startswith('8')) and len(s) == 11:
        return '+' + s[:2] + s[2:]
    
    # Если номер начинается с '7' или '8' и его длина равна 10 — вернуть его в каноническом виде
    if (s.startswith('7') or s.startswith('8')) and len(s) == 10:
        return '+' + s[:1] + s[1:]
    
    # Если номер не начинается с '7' или '8' и его длина равна 11 — вернуть None
    if len(s) == 11 and not (s.startswith('7') or s.startswith('8')):
        return None
    
    # Если номер не начинается с '7' или '8' и его длина равна 10 — вернуть его в каноническом виде
    if len(s) == 10:
        return '+' + s[:1] + s[1:]
    
    # Если все проверки провалились — вернуть None
    return None
