def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Объединяет пересекающиеся и соприкасающиеся отрезки и возвращает их отсортированными по левой границе.

    :param intervals: Список списков с целыми числами, представляющий отрезки.
    :return: Отсортированный список объединенных отрезков.
    """
    # Если список пуст или содержит один элемент, вернуть его как есть
    if len(intervals) <= 1:
        return intervals

    # Сортируем отрезки по левой границе
    intervals.sort(key=lambda x: x[0])

    # Инициализируем результат с первым отрезком
    merged = [intervals[0]]

    # Обрабатываем остальные отрезки
    for current in intervals[1:]:
        # Если текущий отрезок соприкасается или пересекает последний в результате, объединяем их
        if current[0] <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], current[1])
        else:
            # Иначе добавляем текущий отрезок в результат
            merged.append(current)

    return merged
