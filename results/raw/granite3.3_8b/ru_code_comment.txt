def нормализовать_телефон(s: str) -> str | None:
    """
    Нормализует российский номер телефона в каноническое видимое формат '+7XXXXXXXXX'.

    Аргументы:
        s (str): Телефонный номер в произвольном виде.

    Возвращает:
        str | None: Нормализованный телефонный номер или None, если номер не может быть нормализован.
    """
    import re
    
    # Регулярное выражение для совпадения российского номера телефона
    pattern = re.compile(r'^(\+7|8)\s*\(?\d{3}\)?[\s.-]?\d{3}[\-.]?\d{4}$')
    
    # Проверяем, соответствует ли строка регулярному выражению
    match = pattern.match(s)
    
    if not match:
        return None
    
    # Извлекаем группу цифр и добавляем '+' если она отсутствует
    phone_number = ''.join(match.groups())[0]
    if phone_number[0] == '8':
        phone_number = '+' + phone_number[1:]
    
    return phone_number
