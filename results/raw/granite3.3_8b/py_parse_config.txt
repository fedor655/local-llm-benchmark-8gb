import re

def parse_config(text: str) -> dict:
    config = {}
    current_section = None
    pairs = text.strip().split('\n')

    for pair in pairs:
        pair = pair.strip()
        if not pair or pair.lstrip()[0] in ('#', ';'):
            continue

        match = re.match(r'^\s*(\S+)\s*=\s*(.*)', pair)
        if match:
            key, value = match.groups()
            config = _parse_value(config, current_section, key, value)
        elif pair.startswith('[') and pair.endswith(']'):
            current_section = pair[1:-1]
            if current_section not in config:
                config[current_section] = {}
        else:
            if current_section and not current_section in config:
                config[current_section] = {}
            key = '' if not current_section else current_section
            config = _parse_value(config, key, pair, None)

    return config

def _parse_value(config, section, key, value):
    if key in config[section]:
        # Overwrite with the new value
        config[section][key] = _parse_value_from_string(value)
    else:
        config[section].update({key: _parse_value_from_string(value)})

    return config

def _parse_value_from_string(value):
    value = value.strip()
    if value.lower() in ('true', 'false'):
        return value.lower() == 'true'
    if re.match(r'^-?\d+$', value):
        return int(value)
    if re.match(r'^-?\d+\.\d+$', value):
        return float(value)
    if (value.count('"') == 2 and value.startswith('"') and value.endswith('"')):
        value = value[1:-1]
    return value
