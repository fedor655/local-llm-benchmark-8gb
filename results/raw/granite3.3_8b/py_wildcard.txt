def is_match(s: str, p: str) -> bool:
    m, n = len(s), len(p)
    dp = [[False] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = True
    if n > 0 and p[0] == '*':
        dp[0][1] = True

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if p[j - 1] == '*' and (s[i - 1] == p[j - 2] or p[j - 2] == '.'):
                dp[i][j] = dp[i - 1][j] or dp[i][j - 2]
            elif p[j - 1] != '?' and p[j - 1] != '.' and p[j - 1] == s[i - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = dp[i][j - 1]

    return dp[-1][-1]
