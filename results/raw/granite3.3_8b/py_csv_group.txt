                groups[group].append(float(row[value_col]))
            except ValueError:
                pass
    
    def sum_values(group: str) -> float:
        return sum(val for val in groups[group] if val is not None)
    
    sorted_groups = sorted(
        ((group, sum_values(group)) for group in groups

    return [(group, value) for group, value in sorted_groups]
