 ```python
import re

def нормализовать_телефон(s: str) -> str | None:
    """
    Принимает российский номер телефона в произвольном виде и возвращает его в каноническом виде '+79123456789'.
    Если из строки нельзя получить корректный российский номер, вернуть None.
    """
    # Удаляем все ненужные символы и пробелы
    s = re.sub(r'[\-\s\(\)]', '', s)
    
    # Проверяем длину строки после очистки
    if len(s) == 10:
        if s[0] in '89':
            return '+7' + s
    elif len(s) == 11 and s.startswith('7'):
        return '+7' + s[1:]
    
    # Если строка не соответствует ни одному из условий, возвращаем None
    return None
```
