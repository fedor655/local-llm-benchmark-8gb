def search_range(nums, target):
    # Возвращает (первый_индекс, последний_индекс) вхождения target
    # в отсортированном по возрастанию списке nums, либо (-1, -1).
    if not nums:
        return (-1, -1)
    
    def find_first(nums, target):
        lo, hi = 0, len(nums) - 1
        while lo <= hi:
            mid = (lo + hi) // 2
            if nums[mid] < target:
                lo = mid + 1
            else:
                hi = mid - 1
        return lo if lo < len(nums) and nums[lo] == target else -1
    
    def find_last(nums, target):
        lo, hi = 0, len(nums) - 1
        while lo <= hi:
            mid = (lo + hi) // 2
            if nums[mid] <= target:
                lo = mid + 1
            else:
                hi = mid - 1
        return hi if hi >= 0 and nums[hi] == target else -1
    
    first = find_first(nums, target)
    last = find_last(nums, target)
    return (first, last)
