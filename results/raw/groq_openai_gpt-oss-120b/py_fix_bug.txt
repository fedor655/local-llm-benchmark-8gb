def search_range(nums, target):
    # Find first occurrence
    lo, hi = 0, len(nums)
    while lo < hi:
        mid = (lo + hi) // 2
        if nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid
    first = lo
    if first >= len(nums) or nums[first] != target:
        return (-1, -1)

    # Find position after last occurrence
    lo, hi = 0, len(nums)
    while lo < hi:
        mid = (lo + hi) // 2
        if nums[mid] <= target:
            lo = mid + 1
        else:
            hi = mid
    last = lo - 1

    return (first, last)
