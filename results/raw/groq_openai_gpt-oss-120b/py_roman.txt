def int_to_roman(n: int) -> str:
    if not 1 <= n <= 3999:
        raise ValueError("Argument must be between 1 and 3999")
    vals = [
        (1000, "M"), (900, "CM"), (500, "D"), (400, "CD"),
        (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
        (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")
    ]
    res = []
    for value, numeral in vals:
        count, n = divmod(n, value)
        res.append(numeral * count)
    return "".join(res)


def roman_to_int(s: str) -> int:
    if not s:
        raise ValueError("Empty string is not a valid Roman numeral")
    roman_map = {
        'I': 1, 'V': 5, 'X': 10, 'L': 50,
        'C': 100, 'D': 500, 'M': 1000
    }
    total = 0
    prev_value = 0
    for ch in reversed(s.upper()):
        if ch not in roman_map:
            raise ValueError(f"Invalid Roman numeral character: {ch}")
        value = roman_map[ch]
        if value < prev_value:
            total -= value
        else:
            total += value
            prev_value = value
    # Validate by re-encoding to ensure canonical form
    if int_to_roman(total) != s.upper():
        raise ValueError("Invalid or non-canonical Roman numeral")
    return total
