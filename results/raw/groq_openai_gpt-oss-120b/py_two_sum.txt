```python
from collections import defaultdict, deque
from typing import List, Tuple, Optional

def two_sum(nums: List[int], target: int) -> Optional[Tuple[int, int]]:
    # map each value to a queue of its indices (ascending)
    idx_map: defaultdict[int, deque[int]] = defaultdict(deque)
    for idx, val in enumerate(nums):
        idx_map[val].append(idx)

    for i, val in enumerate(nums):
        # remove current index from its queue
        idx_map[val].popleft()
        complement = target - val
        if idx_map[complement]:
            j = idx_map[complement][0]  # smallest index > i
            return (i, j)
    return None
```
