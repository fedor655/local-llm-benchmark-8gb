import re

def нормализовать_телефон(s: str) -> str | None:
    """
    Приводит телефонный номер к каноническому виду +79123456789.

    Принимает строку, содержащую российский номер телефона в произвольном формате.
    Извлекает все цифры и проверяет:
    - 11 цифр, первая из которых 7 или 8 (код страны);
    - 10 цифр без кода страны.
    В случае 11‑значного номера первая цифра заменяется на 7,
    а в случае 10‑значного номера к нему добавляется префикс 7.
    Возвращает строку в виде '+7' + 10 цифр или None, если номер не соответствует требованиям.
    """
    digits = re.sub(r'\D', '', s)
    if len(digits) == 11 and digits[0] in ('7', '8'):
        core = digits[1:]
    elif len(digits) == 10:
        core = digits
    else:
        return None
    return f'+7{core}'
