from typing import List

def merge_intervals(intervals: List[List[int]]) -> List[List[int]]:
    # Make a shallow copy to avoid mutating the input
    sorted_intervals = sorted((interval[:] for interval in intervals), key=lambda x: x[0])
    merged: List[List[int]] = []
    for cur in sorted_intervals:
        if not merged or cur[0] > merged[-1][1]:
            merged.append(cur)
        else:
            merged[-1][1] = max(merged[-1][1], cur[1])
    return merged
