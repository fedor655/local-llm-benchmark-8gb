def parse_config(text: str) -> dict:
    result = {}
    current = None
    pre_section_has_pairs = False

    for raw_line in text.splitlines():
        line = raw_line.strip()
        if not line or line.startswith('#') or line.startswith(';'):
            continue

        if line.startswith('[') and line.endswith(']'):
            current = line[1:-1].strip()
            if current not in result:
                result[current] = {}
            continue

        # key=value line
        if '=' not in line:
            continue
        key, value = line.split('=', 1)
        key = key.strip()
        value = value.strip()

        # ensure a section exists
        if current is None:
            current = ''
            if '' not in result:
                result[''] = {}
            pre_section_has_pairs = True

        # value parsing
        if len(value) >= 2 and value[0] == '"' and value[-1] == '"':
            parsed = value[1:-1]
        else:
            low = value.lower()
            if low == 'true':
                parsed = True
            elif low == 'false':
                parsed = False
            else:
                try:
                    parsed = int(value)
                except ValueError:
                    try:
                        parsed = float(value)
                    except ValueError:
                        parsed = value

        result[current][key] = parsed

    # remove empty pre-section if never used
    if not pre_section_has_pairs and '' in result:
        del result['']

    return result
