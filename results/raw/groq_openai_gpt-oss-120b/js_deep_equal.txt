function deepEqual(a, b) {
  // Fast path for identical values, handling NaN and +/-0 correctly
  if (Object.is(a, b)) return true;

  // If either is null or not an object, they are not equal (primitives already handled)
  if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object')
    return false;

  // Date objects: compare timestamps
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() === b.getTime();
  }

  // Arrays: must both be arrays, same length, and deep-equal elements
  const aIsArr = Array.isArray(a);
  const bIsArr = Array.isArray(b);
  if (aIsArr || bIsArr) {
    if (!aIsArr || !bIsArr) return false;
    if (a.length !== b.length) return false;
    for (let i = 0; i < a.length; i++) {
      if (!deepEqual(a[i], b[i])) return false;
    }
    return true;
  }

  // Plain objects: ensure both are plain objects (not Date, Array, etc.)
  const aTag = Object.prototype.toString.call(a);
  const bTag = Object.prototype.toString.call(b);
  if (aTag !== '[object Object]' || bTag !== '[object Object]') return false;

  const aKeys = Object.keys(a);
  const bKeys = Object.keys(b);
  if (aKeys.length !== bKeys.length) return false;

  // Same set of keys (order doesn't matter)
  for (let key of aKeys) {
    if (!b.hasOwnProperty(key) || !deepEqual(a[key], b[key])) return false;
  }

  return true;
}
