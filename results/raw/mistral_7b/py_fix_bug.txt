def search_range(nums, target):
    # Возвращает (первый_индекс, последний_индекс) вхождения target
    # в отсортированном по возрастанию списке nums, либо (-1, -1).
    lo, hi = 0, len(nums)
    while lo < hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            if mid > 0 and nums[mid-1] == target:
                hi = mid
            else:
                lo = mid + 1
        elif nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid
    first = lo - 1 if lo > 0 and nums[lo-1] == target else -1
    last = next((i for i in range(first, len(nums)) if nums[i] == target), -1)
    return (first, last)
