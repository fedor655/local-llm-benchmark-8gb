def is_match(s, p):
    dp = [[False] * len(p) for _ in range(len(s) + 1)]
    dp[-1][-1] = True

    for i in range(len(s), -1, -1):
        for j in range(len(p) - 1, -1, -1):
            if p[j] == '.':
                dp[i][j] = dp[i + 1][j + 1]
            elif p[j] == '*':
                dp[i][j] = dp[i][j + 1] or (s[i] == p[j - 1] and dp[i + 1][j])
            else:
                dp[i][j] = s[i] == p[j] and dp[i + 1][j + 1]
    return dp[0][0]
