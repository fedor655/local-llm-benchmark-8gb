 ```python
def нормализовать_телефон(s: str) -> str | None:
    import re

    pattern = r'(\+\d{3}|\d{3})[- .]?\d{3}[- .]?\d{2,4}'
    match = re.search(pattern, s)
    if not match:
        return None

    number = match.group(0).replace('+', '').replace('-', '').replace('.', '')
    if len(number) == 10:
        number = '8' + number
    return number
```
