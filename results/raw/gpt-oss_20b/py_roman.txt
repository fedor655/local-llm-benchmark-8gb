def int_to_roman(n: int) -> str:
    if not 1 <= n <= 3999:
        raise ValueError("Number out of range (1..3999)")
    vals = [
        (1000, "M"), (900, "CM"), (500, "D"), (400, "CD"),
        (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
        (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")
    ]
    res = []
    for val, sym in vals:
        while n >= val:
            res.append(sym)
            n -= val
    return "".join(res)

def roman_to_int(s: str) -> int:
    s = s.upper()
    roman_map = {"I": 1, "V": 5, "X": 10, "L": 50,
                 "C": 100, "D": 500, "M": 1000}
    subtractive = {"IV": 4, "IX": 9, "XL": 40, "XC": 90,
                   "CD": 400, "CM": 900}
    i = 0
    total = 0
    while i < len(s):
        if i + 1 < len(s) and s[i:i+2] in subtractive:
            total += subtractive[s[i:i+2]]
            i += 2
        else:
            if s[i] not in roman_map:
                raise ValueError(f"Invalid Roman numeral character: {s[i]}")
            total += roman_map[s[i]]
            i += 1
    return total
