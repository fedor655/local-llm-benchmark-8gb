def is_match(s: str, p: str) -> bool:
    n, m = len(s), len(p)
    # dp[j] will represent dp[i][j] for current i
    dp = [False] * (m + 1)
    dp[0] = True  # empty pattern matches empty string

    # Initialize first row (i = 0, empty string)
    for j in range(1, m + 1):
        if p[j - 1] == '*':
            dp[j] = dp[j - 1]
        else:
            break

    for i in range(1, n + 1):
        prev = dp[0]
        dp[0] = False  # non-empty string cannot match empty pattern
        for j in range(1, m + 1):
            temp = dp[j]
            if p[j - 1] == '*':
                dp[j] = dp[j] or dp[j - 1]  # '*' matches empty or one more char
            elif p[j - 1] == '?' or p[j - 1] == s[i - 1]:
                dp[j] = prev
            else:
                dp[j] = False
            prev = temp
    return dp[m]
