def top_by_group(csv_text: str, group_col: str, value_col: str, k: int) -> list[tuple[str, float]]:
    lines = csv_text.splitlines()
    while lines and lines[-1].strip() == '':
        lines.pop()
    if not lines:
        return []
    header = lines[0].split(',')
    try:
        g_idx = header.index(group_col)
        v_idx = header.index(value_col)
    except ValueError:
        return []
    sums: dict[str, float] = {}
    for line in lines[1:]:
        if not line:
            continue
        parts = line.split(',')
        if len(parts) <= max(g_idx, v_idx):
            continue
        group = parts[g_idx]
        val_str = parts[v_idx]
        if val_str == '':
            continue
        try:
            val = float(val_str)
        except ValueError:
            continue
        sums[group] = sums.get(group, 0.0) + val
    items = list(sums.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:k]
