def two_sum(nums: list[int], target: int) -> tuple[int, int] | None:
    index_of = {}
    best = None
    for j, val in enumerate(nums):
        comp = target - val
        if comp in index_of:
            i = index_of[comp]
            if best is None or i < best[0] or (i == best[0] and j < best[1]):
                best = (i, j)
        if val not in index_of:
            index_of[val] = j
    return best
